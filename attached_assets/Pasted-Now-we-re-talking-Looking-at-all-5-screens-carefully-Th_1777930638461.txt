// Calculate trend: compare avg of first half of period vs second half
const midpoint = Math.floor(dataPoints.length / 2);
const firstHalfAvg = average(dataPoints.slice(0, midpoint));

const trendDiff = secondHalfAvg - firstHalfAvg;
const trendSymbol = trendDiff > 3 ? '↑' : trendDiff < -3 ? '↓' : '—';
